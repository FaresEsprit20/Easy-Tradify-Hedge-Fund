## 1. DOCKER COMPOSE FILES

### `docker-compose.yml`

```yaml
version: '3.8'

services:
  # ============================================================
  # POSTGRESQL DATABASE
  # ============================================================

  postgres:
    image: postgres:15-alpine
    container_name: easytradify-postgres
    environment:
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: root
      POSTGRES_DB: easytradify
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U postgres"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - easytradify-network
    restart: unless-stopped

  # ============================================================
  # DISCOVERY SERVICE
  # ============================================================

  discovery-service:
    build:
      context: ./discovery-server
      dockerfile: Dockerfile
    container_name: easytradify-discovery
    ports:
      - "8761:8761"
    environment:
      - SPRING_PROFILES_ACTIVE=docker
    healthcheck:
      test: ["CMD", "wget", "--spider", "-q", "http://localhost:8761/actuator/health"]
      interval: 30s
      timeout: 10s
      retries: 5
    networks:
      - easytradify-network
    restart: unless-stopped

  # ============================================================
  # EXECUTION SERVICE
  # ============================================================

  execution-service:
    build:
      context: ./execution-service
      dockerfile: Dockerfile
    container_name: easytradify-execution
    ports:
      - "8081:8081"
    environment:
      - SPRING_PROFILES_ACTIVE=docker
      - SPRING_DATASOURCE_URL=jdbc:postgresql://postgres:5432/easytradify-execution-service
      - EUREKA_CLIENT_SERVICEURL_DEFAULTZONE=http://discovery-service:8761/eureka/
      - PYTHON_EXECUTION_URL=http://host.docker.internal:5000
      - INTERNAL_API_KEY=easy-tradify-internal-secret-key-2026
    depends_on:
      postgres:
        condition: service_healthy
      discovery-service:
        condition: service_healthy

      test: ["CMD", "wget", "--spider", "-q", "http://localhost:8081/actuator/health"]
      interval: 30s
      timeout: 10s
      retries: 5

      - easytradify-network

      - "host.docker.internal:host-gateway"

  # MONITOR SERVICE
```

### `docker-build.sh` (Linux/Mac)

```bash
#!/bin/bash

echo "=========================================="
echo "🚀 Building EasyTradify Microservices"
echo "=========================================="

# Build all modules
echo "📦 Building JARs..."
mvn clean package -DskipTests

if [ $? -ne 0 ]; then
    echo "❌ Build failed!"
    exit 1
fi

# Build Docker images
echo "🐳 Building Docker images..."
docker-compose build

echo "=========================================="
echo "✅ Build complete!"
echo ""
echo "📦 To start all services:"
echo "   docker-compose up -d"
echo ""
echo "📦 To view logs:"
echo "   docker-compose logs -f"
echo ""
echo "📦 To stop all services:"
echo "   docker-compose down"
echo ""
echo "📦 To stop and remove volumes:"
echo "   docker-compose down -v"
echo "=========================================="
```

```yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: postgres
  namespace: easytradify
spec:
  replicas: 1
  selector:
    matchLabels:
      app: postgres
  template:
    metadata:
      labels:
        app: postgres
    spec:
      containers:
      - name: postgres
        image: postgres:15-alpine
        env:
        - name: POSTGRES_USER
          value: postgres
        - name: POSTGRES_PASSWORD
          value: root
        - name: POSTGRES_DB
          value: easytradify
        ports:
        - containerPort: 5432
        volumeMounts:
        - name: postgres-storage
          mountPath: /var/lib/postgresql/data
      volumes:
      - name: postgres-storage
        persistentVolumeClaim:
          claimName: postgres-pvc
---
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: postgres-pvc
  namespace: easytradify
spec:
  accessModes:
    - ReadWriteOnce
  resources:
    requests:
      storage: 10Gi
---
apiVersion: v1
kind: Service
metadata:
  name: postgres-service
  namespace: easytradify
spec:
  selector:
    app: postgres
  ports:
    - port: 5432
      targetPort: 5432
```
